Print Lines

Uses the printline function. You can remove lines if you don't want them.

org 0xe9e0

setup:
    setlr
    setsfr
    buffer_clear

text:
    xr0 = 0x3001, adr_of msg1
    printline

    xr0 = 0x3011, adr_of msg2
    printline

    xr0 = 0x3021, adr_of msg3
    printline

    xr0 = 0x3031, adr_of msg4
    printline

    render.ddd4

msg1:
    str "~~~~~Line~1~~~~~~"
    hex 00

msg2:
    str "~~~~~Line~2~~~~~~"
    hex 00

msg3:
    str "~~~~~Line~3~~~~~~"
    hex 00

msg4:
    str "~~~~~Line~4~~~~~~"
    hex 00
